# Commands to upload TurnIT to GitHub
# Run these commands in PowerShell from the TurnIT_code directory

# Step 1: Initialize Git repository (if not already done)
git init

# Step 2: Configure Git (replace with your info)
git config user.name "putbullet"
git config user.email "your-email@example.com"

# Step 3: Add all files
git add .

# Step 4: Check what will be committed
git status

# Step 5: Create initial commit
git commit -m "Initial commit - TurnIT v1.0.0 with bootstrapper"

# Step 6: Add remote repository
git remote add origin https://github.com/putbullet/TurnIT.git

# Step 7: Push to GitHub (main branch)
git push -u origin main

# If the above fails because remote has some content, use force push:
# git push -u origin main --force

# Step 8: Verify upload
# Go to: https://github.com/putbullet/TurnIT

# ============================================================================
# After pushing, create a GitHub Release:
# ============================================================================

# 1. Build the bootstrapper first:
#    .\build_bootstrapper.bat

# 2. Go to: https://github.com/putbullet/TurnIT/releases/new

# 3. Fill in:
#    - Tag: v1.0.0
#    - Title: TurnIT v1.0.0 - Initial Release
#    - Description: See GITHUB_UPLOAD_README.md for release notes template

# 4. Upload: dist\TurnIT-Setup.exe

# 5. Publish release

# ============================================================================
# Subsequent updates:
# ============================================================================

# After making changes to your code:
git add .
git commit -m "Description of changes"
git push

# Users run TurnIT-Setup.exe and automatically get the latest code!
